/* ===================================================== UPLOAD + DELETE GITHUB IMAGE ===================================================== */ /* ===================================================== UPLOAD IMAGE TO CLOUDFLARE WORKER ===================================================== */ async function uploadImageToWorker(file) { try { const formData = new FormData(); /* ================= ACTION ================= */ formData.append( "action", "upload" ); /* ================= IMAGE ================= */ formData.append( "image", file ); /* ================= SEND TO WORKER ================= */ const response = await fetch( WORKER_URL, { method: "POST", body: formData } ); /* ================= READ RESPONSE ================= */ let result; try { result = await response.json(); } catch { throw new Error( "Worker returned invalid response." ); } /* ================= CHECK RESPONSE ================= */ if ( !response.ok || !result.success ) { console.error( "UPLOAD ERROR:", result ); throw new Error( result.message || "Image upload failed." ); } /* ================= SUCCESS ================= */ console.log( "✅ GitHub image uploaded:", result.fileName ); console.log( "🔗 Image URL:", result.image ); return result; } catch (error) { console.error( "❌ IMAGE UPLOAD ERROR:", error ); throw error; } } /* ===================================================== DELETE IMAGE FROM GITHUB ===================================================== */ async function deleteImageFromGitHub( imageName ) { try { /* ================= CHECK ================= */ if (!imageName) { console.log( "ℹ️ No GitHub image name found." ); return { success: true }; } /* ================= FORM DATA ================= */ const formData = new FormData(); /* ================= ACTION ================= */ formData.append( "action", "delete" ); /* ================= IMAGE NAME ================= */ formData.append( "imageName", imageName ); /* ================= SEND TO WORKER ================= */ const response = await fetch( WORKER_URL, { method: "POST", body: formData } ); /* ================= READ RESPONSE ================= */ let result; try { result = await response.json(); } catch { throw new Error( "Worker returned invalid delete response." ); } /* ================= CHECK RESPONSE ================= */ if ( !response.ok || !result.success ) { console.error( "GITHUB DELETE ERROR:", result ); throw new Error( result.message || "GitHub image delete failed." ); } /* ================= SUCCESS ================= */ console.log( "✅ GitHub image deleted:", imageName ); return result; } catch (error) { console.error( "❌ GITHUB IMAGE DELETE ERROR:", error ); throw error; } }
