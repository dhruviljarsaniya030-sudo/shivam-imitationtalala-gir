/* ===================================================== UPLOAD IMAGE TO CLOUDFLARE WORKER ===================================================== */ async function uploadImageToWorker(file) { const formData = new FormData(); /* ================= ACTION ================= */ formData.append( "action", "upload" ); /* ================= IMAGE ================= */ formData.append( "image", file ); /* ================= SEND TO WORKER ================= */ const response = await fetch( WORKER_URL, { method: "POST", body: formData } ); /* ================= RESPONSE ================= */ let result; try { result = await response.json(); } catch (error) { console.error( "WORKER RESPONSE ERROR:", error ); throw new Error( "Worker returned invalid response." ); } /* ================= CHECK RESULT ================= */ if ( !response.ok || !result.success ) { console.error( "IMAGE UPLOAD ERROR:", result ); throw new Error( result.message || "Image upload failed." ); } /* ================= SUCCESS ================= */ console.log( "IMAGE UPLOAD SUCCESS:", result ); return result; }
